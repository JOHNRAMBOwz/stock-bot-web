# === 模块二：9:25 核心狙击系统 ===
st.header("⚡ 模块二：9:25 集合竞价『弱转强』狙击")
st.info("💡 操作指南：请在交易日早上 9:25 分 05秒 之后点击下方按钮，抓取竞价抢筹龙头。")

if st.button("🔫 启动 9:25 终极选股策略"):
    with st.spinner("Stock-Bot 正在全网扫描超预期『弱转强』标的..."):
        try:
            # 1. 抓昨天涨停
            df_zt = ak.stock_zt_pool_em()
            # 筛选昨天换手 10%-30%，剔除 ST和科创创业
            df_zt = df_zt[(df_zt['换手率'] >= 10) & (df_zt['换手率'] <= 30)]
            df_zt = df_zt[~df_zt['名称'].str.contains('ST')]
            df_zt = df_zt[~df_zt['代码'].str.startswith('688')]
            df_zt = df_zt[~df_zt['代码'].str.startswith('300')]
            target_codes = df_zt['代码'].tolist()

            if not target_codes:
                st.warning("昨日无符合换手率条件的股票，今日空仓。")
            else:
                # 2. 抓今日开盘快照
                df_spot = ak.stock_zh_a_spot_em()
                df_match = df_spot[df_spot['代码'].isin(target_codes)]
                
                results = []
                for index, row in df_match.iterrows():
                    code = row['代码']
                    name = row['名称']
                    open_price = row['今开']
                    prev_close = row['昨收']
                    
                    if open_price == 0 or pd.isna(open_price) or prev_close == 0:
                        continue
                        
                    gap_up_pct = (open_price - prev_close) / prev_close * 100
                    lianban = df_zt[df_zt['代码'] == code]['连板数'].values[0]
                    huanshou = df_zt[df_zt['代码'] == code]['换手率'].values[0]

                    # 3% 到 9% 的高开
                    if 3.0 <= gap_up_pct <= 9.0:
                        results.append({
                            '股票代码': code,
                            '股票名称': name,
                            '昨日连板': lianban,
                            '昨日换手(%)': round(huanshou, 2),
                            '今日高开(%)': round(gap_up_pct, 2)
                        })
                
                if results:
                    st.success(f"🎉 狙击成功！Stock-Bot 发现 {len(results)} 只符合黄金买点的标的！")
                    df_res = pd.DataFrame(results)
                    st.dataframe(df_res, use_container_width=True)
                else:
                    st.error("😭 竞价结束，未发现符合 3%-9% 高开条件的股票。执行纪律：管住手！")
                    
        except Exception as e:
            st.error(f"运行报错：{e}")
